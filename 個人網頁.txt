<script>
  const intro = document.getElementById("intro");
  const site = document.getElementById("site");
  const mainText = document.getElementById("mainText");
  const stage = document.getElementById("textStage");
  const strobe = document.getElementById("strobe");
  const slice = document.getElementById("slice");
  const personFlash = document.getElementById("personFlash");

  const glitchSound = document.getElementById("glitchSound");
  const enterSound = document.getElementById("enterSound");

  glitchSound.volume = 0.15;
  enterSound.volume = 0.12;

  function playGlitch(){
    glitchSound.currentTime = 0;
    glitchSound.play().catch(()=>{});
  }

  function playEnter(){
    enterSound.currentTime = 0;
    enterSound.play().catch(()=>{});
  }

  function flashFrame(){
    strobe.style.opacity="1";
    setTimeout(()=> strobe.style.opacity="0", 16);
  }

  function flashPerson(){
    personFlash.style.opacity="1";
    setTimeout(()=> personFlash.style.opacity="0", 16);
  }

  function tearSlice(){
    slice.style.opacity="1";
    slice.style.top = (30 + Math.random()*40) + "%";
    slice.style.height = (8 + Math.random()*22) + "px";
    slice.style.transform =
      `translateX(${(Math.random()>0.5?"-":"")}${80+Math.random()*240}px)`;
    setTimeout(()=> slice.style.opacity="0", 50);
  }

  function createGhost(offsetX, offsetY, opacity, blur, color, stretch){
    const g = document.createElement("div");
    g.className="ghost " + color;
    g.textContent="AKUMA";
    g.style.opacity=opacity;
    g.style.filter= blur ? `blur(${blur}px)`:"none";

    const jx = Math.floor(Math.random()*20)-10;
    const jy = Math.floor(Math.random()*14)-7;
    const scaleX = stretch ? (1.2+Math.random()*0.3) : 1;

    g.style.transform =
      `translate(calc(-50% + ${offsetX+jx}px),
                 calc(-50% + ${offsetY+jy}px))
                 scaleX(${scaleX})`;

    stage.appendChild(g);

    setTimeout(()=>{
      g.style.opacity = Math.random()>0.5? opacity : 0;
    }, 30+Math.random()*80);

    setTimeout(()=> g.remove(), 200+Math.random()*200);
  }

  function heavyBurst(){
    const count = 20 + Math.floor(Math.random()*21);

    for(let i=0;i<count;i++){
      const leftTrail = -(40 + Math.random()*260);
      const y = (Math.random()*30)-15;

      const opacity = 0.1 + Math.random()*0.6;
      const blur = Math.random()>0.4 ? 0.5+Math.random()*3 : 0;

      let color="white";
      const r=Math.random();
      if(r>0.66) color="red";
      else if(r>0.33) color="blue";

      const stretch = Math.random()>0.6;

      createGhost(leftTrail,y,opacity,blur,color,stretch);
    }

    if(Math.random()>0.4) tearSlice();

    if(Math.random()>0.72){
      flashFrame();
      playGlitch();
    }

    if(Math.random()>0.82){
      flashPerson();
    }
  }

  /* =========================
     INTRO SEQUENCE
  ========================= */

  // first flash
  setTimeout(()=> {
    flashFrame();
    playGlitch();
  }, 40);

  // slide in sync sound
  setTimeout(()=>{
    mainText.style.opacity="1";
    mainText.style.transition="transform 0.12s linear";
    mainText.style.transform="translateX(60px)";
    playGlitch(); // sync
  }, 120);

  // snap back
  setTimeout(()=>{
    mainText.style.transition="transform 0.05s linear";
    mainText.style.transform="translateX(0)";
  }, 260);

  // hold clean text (0.5 sec)
  setTimeout(()=>{
    mainText.style.opacity="1";
    mainText.style.transform="translate(0,0)";
    mainText.style.letterSpacing="-2px";
  }, 320);

  // explosion start after hold
  setTimeout(()=>{
    const glitchInterval = setInterval(()=>{

      heavyBurst();

      const jx=Math.floor(Math.random()*12)-6;
      const jy=Math.floor(Math.random()*12)-6;
      mainText.style.transform=`translate(${jx}px,${jy}px)`;

      mainText.style.opacity=Math.random()>0.2?"1":"0";

      if(Math.random()>0.7){
        mainText.style.letterSpacing = Math.random()>0.5 ? "6px" : "-2px";
      }

    }, 50);

    setTimeout(()=>{
      clearInterval(glitchInterval);
      mainText.style.opacity="1";
      mainText.style.transform="translate(0,0)";
      mainText.style.letterSpacing="-2px";
    }, 900);

  }, 820);

  // final burst
  setTimeout(()=>{
    for(let i=0;i<4;i++) heavyBurst();
    flashFrame();
    playGlitch();
  }, 1450);

  // cut to site with glitch fade-in
  setTimeout(()=>{
    intro.style.display="none";
    site.classList.add("glitchIn");
    playEnter();
  }, 1750);
</script>
